// These are the Firebase Storage rules you should apply in your Firebase console
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /users/{userId}/{allPaths=**} {
      // Allow read access to anyone
      allow read: if true;
      // Allow write access to authenticated users for their own folder
      allow write: if request.auth != null && request.auth.uid == userId;
    }
    
    // Default rule - deny all other access
    match /{allPaths=**} {
      allow read: if true;
      allow write: if false;
    }
  }
}
