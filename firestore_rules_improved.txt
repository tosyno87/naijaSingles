rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper function to check if two users are matched
    function areUsersMatched(userId1, userId2) {
      return exists(/databases/$(database)/documents/matches/$(userId1 + '_' + userId2)) ||
             exists(/databases/$(database)/documents/matches/$(userId2 + '_' + userId1)) ||
             exists(/databases/$(database)/documents/Matches/$(userId1 + '_' + userId2)) ||
             exists(/databases/$(database)/documents/Matches/$(userId2 + '_' + userId1));
    }

    // Helper function to check if user is blocked
    function isUserBlocked(userId, blockedUserId) {
      return exists(/databases/$(database)/documents/users/$(userId)/blockedlist/$(blockedUserId)) ||
             exists(/databases/$(database)/documents/users/$(blockedUserId)/blockedlist/$(userId));
    }

    // Users collection - allow authenticated users to read others for matching
    match /users/{userId} {
      // Users can read and write their own data
      allow read, write: if request.auth != null && request.auth.uid == userId;
      
      // Authenticated users can read other users' profiles for matching/discovery
      // But not if they're blocked
      allow read: if request.auth != null && 
                   !isUserBlocked(request.auth.uid, userId);
      
      // Allow users to update their own profile with data validation
      allow update: if request.auth != null && 
                     request.auth.uid == userId &&
                     // Prevent users from changing critical fields
                     !request.resource.data.diff(resource.data).affectedKeys()
                       .hasAny(['createdAt', 'uid', 'email']);
      
      // User subcollections
      match /CheckedUser/{checkedUserId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }
      
      match /LikedBy/{likedById} {
        // Allow the user being liked to read who liked them
        allow read: if request.auth != null && request.auth.uid == userId;
        // Allow the liker to write their like
        allow write: if request.auth != null && request.auth.uid == likedById;
        // Allow both users to read the like data
        allow read: if request.auth != null && 
                     (request.auth.uid == userId || request.auth.uid == likedById);
      }
      
      match /Matches/{matchId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }
      
      match /blockedlist/{blockedId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }
    }

    // Likes collection - for the new likes service
    match /likes/{likeId} {
      // Users can create likes where they are the 'from' user and target isn't blocked
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.from &&
                     !isUserBlocked(request.auth.uid, request.resource.data.to);
      
      // Users can read likes where they are involved (from or to)
      allow read: if request.auth != null && 
                   (request.auth.uid == resource.data.from || 
                    request.auth.uid == resource.data.to);
      
      // Users can update likes where they are involved (for mutual likes)
      allow update: if request.auth != null && 
                     (request.auth.uid == resource.data.from || 
                      request.auth.uid == resource.data.to);
      
      // Users can delete their own likes
      allow delete: if request.auth != null && 
                     request.auth.uid == resource.data.from;
    }

    // Legacy likes collection (if exists) - More restrictive
    match /Likes/{likeId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && 
                    request.auth.uid == request.resource.data.from;
    }

    // Matches collection - for storing user matches
    match /matches/{matchId} {
      // Users can read matches they're part of
      allow read: if request.auth != null && 
                   request.auth.uid in resource.data.users;
      
      // Users can create matches involving themselves (usually done by system)
      allow create: if request.auth != null && 
                     request.auth.uid in request.resource.data.users &&
                     request.resource.data.users.size() == 2;
      
      // Users can update matches they're part of (for status changes)
      allow update: if request.auth != null && 
                     request.auth.uid in resource.data.users;
      
      // Users can delete matches they're part of (unmatch)
      allow delete: if request.auth != null && 
                     request.auth.uid in resource.data.users;
    }

    // Legacy matches collection (for backward compatibility)
    match /Matches/{matchId} {
      // Users can read matches they're part of
      allow read: if request.auth != null && 
                   (request.auth.uid in resource.data.users ||
                    request.auth.uid == resource.data.user1 ||
                    request.auth.uid == resource.data.user2);
      
      // Users can create matches involving themselves
      allow create: if request.auth != null && 
                     (request.auth.uid in request.resource.data.users ||
                      request.auth.uid == request.resource.data.user1 ||
                      request.auth.uid == request.resource.data.user2);
      
      // Users can update matches they're part of
      allow update: if request.auth != null && 
                     (request.auth.uid in resource.data.users ||
                      request.auth.uid == resource.data.user1 ||
                      request.auth.uid == resource.data.user2);
    }

    // Chat threads collection - ENHANCED WITH MATCH VERIFICATION
    match /chatThreads/{threadId} {
      // Users can read chat threads they're part of
      allow read: if request.auth != null &&
                   request.auth.uid in resource.data.userIds &&
                   !isUserBlocked(request.auth.uid, 
                     resource.data.userIds[resource.data.userIds[0] == request.auth.uid ? 1 : 0]);
      
      // Users can create chat threads ONLY with matched users
      allow create: if request.auth != null &&
                     request.auth.uid in request.resource.data.userIds &&
                     request.resource.data.userIds.size() == 2 &&
                     areUsersMatched(request.resource.data.userIds[0], request.resource.data.userIds[1]) &&
                     !isUserBlocked(request.auth.uid, 
                       request.resource.data.userIds[request.resource.data.userIds[0] == request.auth.uid ? 1 : 0]);
      
      // Users can update chat threads they're part of (for last message, unread count)
      allow update: if request.auth != null &&
                     request.auth.uid in resource.data.userIds &&
                     // Prevent changing userIds after creation
                     !request.resource.data.diff(resource.data).affectedKeys().hasAny(['userIds']) &&
                     !isUserBlocked(request.auth.uid, 
                       resource.data.userIds[resource.data.userIds[0] == request.auth.uid ? 1 : 0]);
      
      // Users can delete chat threads they're part of
      allow delete: if request.auth != null &&
                     request.auth.uid in resource.data.userIds;
    }

    // Messages subcollection within chat threads - ENHANCED SECURITY
    match /chatThreads/{threadId}/messages/{messageId} {
      // Users can read messages in threads they're part of
      allow read: if request.auth != null && 
                   exists(/databases/$(database)/documents/chatThreads/$(threadId)) &&
                   request.auth.uid in get(/databases/$(database)/documents/chatThreads/$(threadId)).data.userIds;
      
      // Users can create messages in threads they're part of with validation
      allow create: if request.auth != null && 
                     exists(/databases/$(database)/documents/chatThreads/$(threadId)) &&
                     request.auth.uid in get(/databases/$(database)/documents/chatThreads/$(threadId)).data.userIds &&
                     request.resource.data.senderId == request.auth.uid &&
                     // Message validation
                     request.resource.data.keys().hasAll(['senderId', 'text', 'timestamp']) &&
                     request.resource.data.text is string &&
                     request.resource.data.text.size() > 0 &&
                     request.resource.data.text.size() <= 1000; // Max message length
      
      // Users can update messages for read status only
      allow update: if request.auth != null && 
                     exists(/databases/$(database)/documents/chatThreads/$(threadId)) &&
                     request.auth.uid in get(/databases/$(database)/documents/chatThreads/$(threadId)).data.userIds &&
                     // Only allow updating read status, not content
                     request.resource.data.diff(resource.data).affectedKeys().hasOnly(['read']);
      
      // Users can delete their own messages only
      allow delete: if request.auth != null && 
                     exists(/databases/$(database)/documents/chatThreads/$(threadId)) &&
                     request.auth.uid in get(/databases/$(database)/documents/chatThreads/$(threadId)).data.userIds &&
                     resource.data.senderId == request.auth.uid;
    }

    // Typing indicators subcollection (for future real-time typing)
    match /chatThreads/{threadId}/typing/{userId} {
      // Users can read typing status in threads they're part of
      allow read: if request.auth != null && 
                   exists(/databases/$(database)/documents/chatThreads/$(threadId)) &&
                   request.auth.uid in get(/databases/$(database)/documents/chatThreads/$(threadId)).data.userIds;
      
      // Users can only update their own typing status
      allow write: if request.auth != null && 
                    exists(/databases/$(database)/documents/chatThreads/$(threadId)) &&
                    request.auth.uid in get(/databases/$(database)/documents/chatThreads/$(threadId)).data.userIds &&
                    request.auth.uid == userId;
    }

    // Reports collection - for user reports with validation
    match /reports/{reportId} {
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.reporterId &&
                     request.resource.data.keys().hasAll(['reporterId', 'reportedUserId', 'reason', 'timestamp']) &&
                     request.resource.data.reason is string &&
                     request.resource.data.reason.size() > 0;
      allow read: if request.auth != null && 
                   request.auth.uid == resource.data.reporterId;
    }

    // Notifications collection with better validation
    match /notifications/{notificationId} {
      allow read: if request.auth != null && 
                   request.auth.uid == resource.data.userId;
      allow write: if request.auth != null && 
                    request.auth.uid == resource.data.userId;
      // System can create notifications for users
      allow create: if request.auth != null;
    }

    // User settings/preferences
    match /userSettings/{userId} {
      allow read, write: if request.auth != null && 
                          request.auth.uid == userId;
    }

    // Item access collection (for app configuration)
    match /Item_access/{itemId} {
      allow read: if request.auth != null;
    }

    // Block any other access not explicitly allowed above
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
