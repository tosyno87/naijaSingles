# Firebase Firestore Security Rules Update
# Add these rules to your firestore.rules file to support new cultural fields

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Users collection rules
    match /users/{userId} {
      // Allow read/write for authenticated users
      allow read, write: if request.auth != null && request.auth.uid == userId;
      
      // Allow read for other users (for discovery) but with privacy controls
      allow read: if request.auth != null && 
        // Only allow reading public cultural fields
        resource.data.keys().hasAll(['nationality', 'tribe', 'languages']) &&
        // Ensure sensitive data is not exposed
        !resource.data.keys().hasAny(['phoneNumber', 'email', 'address']);
      
      // Public profile subcollection
      match /public/{document=**} {
        allow read: if request.auth != null;
        allow write: if request.auth != null && request.auth.uid == userId;
      }
      
      // Private profile subcollection
      match /private/{document=**} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }
    }
    
    // Events collection rules
    match /events/{eventId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && 
        (resource == null || resource.data.organizerId == request.auth.uid);
    }
    
    // Messages collection rules
    match /messages/{messageId} {
      allow read, write: if request.auth != null && 
        (resource.data.senderId == request.auth.uid || 
         resource.data.receiverId == request.auth.uid);
    }
    
    // Reports collection rules
    match /reports/{reportId} {
      allow read: if request.auth != null && 
        resource.data.reporterId == request.auth.uid;
      allow write: if request.auth != null;
    }
    
    // Feedback collection rules
    match /feedback/{feedbackId} {
      allow read, write: if request.auth != null;
    }
  }
}

# Cultural field validation examples:
# - nationality: string, required, max 50 chars
# - tribe: string, required, max 30 chars  
# - languages: array of strings, required, max 10 items
# - religion: string, optional, max 30 chars
# - occupation: string, optional, max 50 chars

# Privacy considerations:
# - nationality and tribe are public for matching
# - languages are public for compatibility
# - religion and occupation are private by default
# - All cultural fields should be validated before saving
